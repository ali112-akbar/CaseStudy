Q1. Do customers from the UK spend more per transaction than customers from Germany?


--- code ---
# Merge transactions with customers on customer_id
df_merged = df_trans.merge(df_cust, on="customer_id", how="left")

# Filter UK and Germany customers
uk_df = df_merged[df_merged["country"] == "UK"]
de_df = df_merged[df_merged["country"] == "Germany"]

# Compute average spend per transaction
uk_avg = uk_df["purchase_value"].mean()
de_avg = de_df["purchase_value"].mean()

print("Average spend per transaction:")
print(f"UK: {uk_avg:.2f}")
print(f"Germany: {de_avg:.2f}")

# Comparison
if uk_avg > de_avg:
    print("UK customers spend more per transaction.")
elif uk_avg < de_avg:
    print("German customers spend more per transaction.")
else:
    print("Both countries spend the same per transaction.")

# Step 1: Compute total spend per customer
customer_spend = df_trans.groupby("customer_id")["purchase_value"].sum().reset_index()

# Step 2: Define high-spending customers (top 20%)
threshold = customer_spend["purchase_value"].quantile(0.80)
customer_spend["high_spender"] = customer_spend["purchase_value"] >= threshold

# Step 3: Merge with web session data
df_merged = df_web.merge(customer_spend, on="customer_id", how="left")

# Step 4: Compute average session time for high vs non-high spenders

high_avg  = df_merged[df_merged["high_spender"] == True]["time_on_site"].mean()
low_avg   = df_merged[df_merged["high_spender"] == False]["time_on_site"].mean()

print("Average session time:")
print(f"High spenders: {high_avg:.2f} minutes")

    print("Higher engagement is associated with lower purchase value.")
else:
    print("Engagement and purchase value have weak or no correlation.")



# Define simple, human-friendly time buckets
bins = [0, 50, 100, 150, 200, df_merged["time_on_site"].max()]
labels = ["0–50", "50–100", "100–150", "150–200", "200+"]

df_merged["time_bucket"] = pd.cut(df_merged["time_on_site"], bins=bins, labels=labels, include_lowest=True)

# Compute average purchase value per bucket
avg_values = df_merged.groupby("time_bucket")["purchase_value"].mean().reset_index()
